    USUARIO {
        bigint id PK
        varchar username UK
        varchar password
        varchar first_name
        varchar last_name
        varchar email
        varchar rol
        varchar telefono
        boolean is_active
    }
    CLIENTE {
        bigint id PK
        varchar nombres
        varchar apellidos
        varchar telefono
        varchar email
        varchar direccion
        varchar nit
        boolean activo
        datetime creado
    }
    MASCOTA {
        bigint id PK
        bigint cliente_id FK
        varchar nombre
        varchar especie
        varchar raza
        char sexo
        date fecha_nacimiento
        varchar color
        boolean esterilizado
        text notas
        boolean activo
    }
    SERVICIO {
        bigint id PK
        varchar nombre UK
        text descripcion
        decimal precio
        smallint duracion_minutos
        boolean activo
    }
    CITA {
        bigint id PK
        bigint mascota_id FK
        bigint veterinario_id FK
        bigint servicio_id FK
        bigint creado_por_id FK
        datetime fecha_hora
        smallint duracion_minutos
        varchar estado
        text motivo
        datetime creado
    }
    CONSULTA {
        bigint id PK
        bigint mascota_id FK
        bigint cita_id FK
        bigint veterinario_id FK
        datetime fecha
        decimal peso_kg
        decimal temperatura_c
        text sintomas
        text diagnostico
        text tratamiento
        text observaciones
        decimal costo
    }
    VACUNA {
        bigint id PK
        bigint mascota_id FK
        bigint aplicada_por_id FK
        varchar nombre
        date fecha_aplicacion
        date proxima_dosis
        varchar lote
        varchar notas
    }
